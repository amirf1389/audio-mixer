# Audio Mixer / A mix OS engine (C++20): the engine library, the runtime (device backends of the system it is built on) and the tests.
#   cmake -S engine/cpp -B build && cmake --build build && ctest --test-dir build --output-on-failure
cmake_minimum_required(VERSION 3.12)
project(audiomixer_engine C CXX)
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/winres.cmake)   # amfb_win_version: publisher / product / version on every Windows binary
set(CMAKE_CXX_STANDARD 20)   # the channel DSP (include/audiomixer/dsp.hpp) is C++20; everything else compiles as before
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)   # the static libs also go into libamix.so (the C SDK)

add_library(audiomixer_engine STATIC src/engine.cpp)
target_include_directories(audiomixer_engine PUBLIC include)

set(RUNTIME_SRC src/runtime.cpp src/backend_null.cpp src/devices.cpp src/driver_intel.cpp src/bluetooth.cpp src/sysopt.cpp src/app_sound.cpp src/latency.cpp)
set(RUNTIME_LIBS Threads::Threads)
if(WIN32)
  list(APPEND RUNTIME_SRC src/backend_wasapi.cpp src/backend_asio.cpp src/backend_dsound.cpp)
  list(APPEND RUNTIME_LIBS ole32 oleaut32 uuid advapi32 avrt dsound winmm bthprops setupapi)
  set(ASIO_DRIVER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../drivers/asio)   # asio_sdk.h + the Steinberg ASIO SDK headers (GPLv3)
elseif(APPLE)
  enable_language(OBJCXX)
  list(APPEND RUNTIME_SRC src/backend_coreaudio.cpp)
  list(APPEND RUNTIME_LIBS "-framework AudioToolbox" "-framework CoreAudio" "-framework AudioUnit")
else()
  find_package(ALSA)
  if(ALSA_FOUND)
    list(APPEND RUNTIME_SRC src/backend_alsa.cpp)
    list(APPEND RUNTIME_LIBS ALSA::ALSA)
    set(RUNTIME_DEFS AM_HAVE_ALSA)
  endif()
endif()
add_library(audiomixer_runtime STATIC ${RUNTIME_SRC})
target_link_libraries(audiomixer_runtime PUBLIC audiomixer_engine amixos_paths ${RUNTIME_LIBS})
if(WIN32)
  target_include_directories(audiomixer_runtime PRIVATE ${ASIO_DRIVER_DIR})
endif()
if(RUNTIME_DEFS)
  target_compile_definitions(audiomixer_runtime PUBLIC ${RUNTIME_DEFS})
endif()

# The native program: the engine on this operating system's own audio layer (engine/cpp/app/main.cpp)
add_executable(audiomixer-native app/main.cpp app/stream.cpp)
if(WIN32)
  target_sources(audiomixer-native PRIVATE app/win_tools.cpp)
  target_link_libraries(audiomixer-native winhttp bcrypt)   # --install-driver / --uninstall-driver / --install-flexasio (the driver DLL is embedded by scripts/build-native.js)
endif()
target_link_libraries(audiomixer-native audiomixer_runtime)
if(WIN32)
  target_include_directories(audiomixer-native PRIVATE ${ASIO_DRIVER_DIR})
  target_link_options(audiomixer-native PRIVATE -static)
endif()

if(WIN32)
  target_link_options(amixos-daemon PRIVATE -static)
  if(MINGW)
    # A toolchain file may leave the linker in dynamic mode (-Bdynamic) for the programs that need Qt's DLLs; the C++ runtime would then
    # come from whatever libstdc++-6.dll lies beside the program, and an older one lacks symbols ("entry point not found"). The daemon
    # carries its own runtime: these go last on the link line.
    target_link_libraries(amixos-daemon -Wl,-Bstatic,-lstdc++,-lwinpthread)
    target_link_libraries(amfb_network PRIVATE -Wl,-Bstatic,-lstdc++,-lwinpthread)
  endif()
endif()
