# Audio Mixer / A mix OS engine (C++20): the engine library, the runtime (device backends of the system it is built on) and the tests.
#   cmake -S engine/cpp -B build && cmake --build build && ctest --test-dir build --output-on-failure
cmake_minimum_required(VERSION 3.12)
project(audiomixer_engine C CXX)
set(CMAKE_CXX_STANDARD 20)   # the channel DSP (include/audiomixer/dsp.hpp) is C++20; everything else compiles as before
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)   # the static libs also go into libamix.so (the C SDK)

add_library(audiomixer_engine STATIC src/engine.cpp)
target_include_directories(audiomixer_engine PUBLIC include)

set(RUNTIME_SRC src/runtime.cpp src/backend_null.cpp src/devices.cpp src/driver_intel.cpp src/bluetooth.cpp)
set(RUNTIME_LIBS Threads::Threads)
if(WIN32)
  list(APPEND RUNTIME_SRC src/backend_wasapi.cpp src/backend_asio.cpp src/backend_dsound.cpp)
  list(APPEND RUNTIME_LIBS ole32 oleaut32 uuid advapi32 avrt dsound winmm bthprops)
  set(ASIO_DRIVER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../drivers/asio)   # asio_sdk.h + the Steinberg ASIO SDK headers (GPLv3)
elseif(APPLE)
  enable_language(OBJCXX)
  list(APPEND RUNTIME_SRC src/backend_coreaudio.cpp)
  list(APPEND RUNTIME_LIBS "-framework AudioToolbox" "-framework CoreAudio" "-framework AudioUnit")
else()
  find_package(ALSA)
  if(ALSA_FOUND)
    list(APPEND RUNTIME_SRC src/backend_alsa.cpp)
    list(APPEND RUNTIME_LIBS ALSA::ALSA)
    set(RUNTIME_DEFS AM_HAVE_ALSA)
  endif()
endif()
add_library(audiomixer_runtime STATIC ${RUNTIME_SRC})
target_link_libraries(audiomixer_runtime PUBLIC audiomixer_engine amixos_paths ${RUNTIME_LIBS})
if(WIN32)
  target_include_directories(audiomixer_runtime PRIVATE ${ASIO_DRIVER_DIR})
endif()
if(RUNTIME_DEFS)
  target_compile_definitions(audiomixer_runtime PUBLIC ${RUNTIME_DEFS})
endif()

# The native program: the engine on this operating system's own audio layer (engine/cpp/app/main.cpp)
add_executable(audiomixer-native app/main.cpp app/stream.cpp)
if(WIN32)
  target_sources(audiomixer-native PRIVATE app/win_tools.cpp)
  target_link_libraries(audiomixer-native winhttp bcrypt)   # --install-driver / --uninstall-driver / --install-flexasio (the driver DLL is embedded by scripts/build-native.js)
endif()
target_link_libraries(audiomixer-native audiomixer_runtime)
if(WIN32)
  target_include_directories(audiomixer-native PRIVATE ${ASIO_DRIVER_DIR})
  target_link_options(audiomixer-native PRIVATE -static)
endif()

# amixos-daemon: the engine as a background service with a local JSON API (engine/cpp/app/daemon). The Kotlin remote (android/) talks to it.
add_library(amixos_daemon_lib STATIC app/daemon/daemon.cpp app/daemon/httpd.cpp app/daemon/tokens.cpp)
target_include_directories(amixos_daemon_lib PUBLIC app)
target_link_libraries(amixos_daemon_lib PUBLIC audiomixer_runtime amixos_license)
if(WIN32)
  target_link_libraries(amixos_daemon_lib PUBLIC ws2_32)
endif()
# the network service as an extension (a shared library the console loads) and as a stand-alone program
add_library(amfb_network_ext STATIC app/daemon/main.cpp)
target_link_libraries(amfb_network_ext PUBLIC amixos_daemon_lib)
set_target_properties(amfb_network_ext PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(amfb_network SHARED app/ext/ext_shared.cpp)
target_link_libraries(amfb_network PRIVATE amfb_network_ext)
set_target_properties(amfb_network PROPERTIES OUTPUT_NAME "AmF X1 Network" PREFIX "")
add_executable(amixos-daemon app/ext/ext_main.cpp)
target_link_libraries(amixos-daemon amfb_network_ext)
if(WIN32)
  target_link_options(amixos-daemon PRIVATE -static)
  if(MINGW)
    # A toolchain file may leave the linker in dynamic mode (-Bdynamic) for the programs that need Qt's DLLs; the C++ runtime would then
    # come from whatever libstdc++-6.dll lies beside the program, and an older one lacks symbols ("entry point not found"). The daemon
    # carries its own runtime: these go last on the link line.
    target_link_libraries(amixos-daemon -Wl,-Bstatic,-lstdc++,-lwinpthread)
    target_link_libraries(amfb_network PRIVATE -Wl,-Bstatic,-lstdc++,-lwinpthread)
  endif()
endif()

# ---- the C SDK: one shared library (and a static twin) over the runtime, the engine and the licence check; header in sdk/c/include
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../native/amixos/version.txt" AMIX_SDK_VERSION)
string(STRIP "${AMIX_SDK_VERSION}" AMIX_SDK_VERSION)
add_library(amix SHARED app/sdk/amix_c.cpp)
add_library(amix_static STATIC app/sdk/amix_c.cpp)
foreach(t amix amix_static)
  target_include_directories(${t} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../../sdk/c/include PRIVATE app)
  target_link_libraries(${t} PUBLIC audiomixer_runtime audiomixer_engine amixos_license amixos_net)
  target_compile_definitions(${t} PRIVATE AMIX_VERSION_STRING="${AMIX_SDK_VERSION}")
endforeach()
set_target_properties(amix PROPERTIES CXX_VISIBILITY_PRESET hidden VERSION ${AMIX_SDK_VERSION} SOVERSION 1 PUBLIC_HEADER ${CMAKE_CURRENT_SOURCE_DIR}/../../sdk/c/include/amix.h)
target_compile_definitions(amix_static PUBLIC AMIX_STATIC)
if(NOT WIN32 AND NOT APPLE)
  target_link_options(amix PRIVATE -Wl,--exclude-libs,ALL)   # only the amix_* C symbols leave the shared library
endif()
add_executable(sdk_c_test test/sdk_c_test.c)
target_link_libraries(sdk_c_test PRIVATE amix m)
add_test(NAME sdk_c_test COMMAND sdk_c_test)
install(TARGETS amix amixos-daemon LIBRARY DESTINATION lib RUNTIME DESTINATION bin PUBLIC_HEADER DESTINATION include)
