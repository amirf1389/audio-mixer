# Audio Mixer engine (C++17): the engine library, the runtime (device backends of the system it is built on) and the tests.
#   cmake -S engine/cpp -B build && cmake --build build && ctest --test-dir build --output-on-failure
cmake_minimum_required(VERSION 3.12)
project(audiomixer_engine CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

add_library(audiomixer_engine STATIC src/engine.cpp)
target_include_directories(audiomixer_engine PUBLIC include)

set(RUNTIME_SRC src/runtime.cpp src/backend_null.cpp src/spatial.cpp)
set(RUNTIME_LIBS Threads::Threads)
if(WIN32)
  list(APPEND RUNTIME_SRC src/backend_wasapi.cpp src/backend_asio.cpp)
  list(APPEND RUNTIME_LIBS ole32 uuid advapi32)
  set(ASIO_DRIVER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../drivers/asio)   # asio_sdk.h + the Steinberg ASIO SDK headers (GPLv3)
elseif(APPLE)
  enable_language(OBJCXX)
  list(APPEND RUNTIME_SRC src/backend_coreaudio.cpp)
  list(APPEND RUNTIME_LIBS "-framework AudioToolbox" "-framework CoreAudio" "-framework AudioUnit")
else()
  find_package(ALSA)
  if(ALSA_FOUND)
    list(APPEND RUNTIME_SRC src/backend_alsa.cpp)
    list(APPEND RUNTIME_LIBS ALSA::ALSA)
    set(RUNTIME_DEFS AM_HAVE_ALSA)
  endif()
endif()
# Optional 360 degree output (Resonance Audio, Apache-2.0): -DAM_RESONANCE_INCLUDE=<resonance_audio/api dir> -DAM_RESONANCE_LIB=<libResonanceAudioStatic.a>
# (node scripts/build-resonance.js fetches and builds the pinned sources). Without them am::Spatial is a pass-through and Spatial::available() is false.
set(AM_RESONANCE_INCLUDE "" CACHE PATH "Resonance Audio API headers (resonance_audio/api)")
set(AM_RESONANCE_LIB "" CACHE FILEPATH "Resonance Audio static library")
add_library(audiomixer_runtime STATIC ${RUNTIME_SRC})
if(AM_RESONANCE_INCLUDE AND AM_RESONANCE_LIB)
  target_include_directories(audiomixer_runtime PRIVATE ${AM_RESONANCE_INCLUDE})
  target_compile_definitions(audiomixer_runtime PRIVATE AM_HAVE_RESONANCE)
  target_link_libraries(audiomixer_runtime PUBLIC ${AM_RESONANCE_LIB})
  set(HAVE_RESONANCE ON)
endif()
target_link_libraries(audiomixer_runtime PUBLIC audiomixer_engine ${RUNTIME_LIBS})
if(WIN32)
  target_include_directories(audiomixer_runtime PRIVATE ${ASIO_DRIVER_DIR})
endif()
if(RUNTIME_DEFS)
  target_compile_definitions(audiomixer_runtime PUBLIC ${RUNTIME_DEFS})
endif()

# The native program: the engine on this operating system's own audio layer (engine/cpp/app/main.cpp)
add_executable(audiomixer-native app/main.cpp)
target_link_libraries(audiomixer-native audiomixer_runtime)
if(WIN32)
  target_include_directories(audiomixer-native PRIVATE ${ASIO_DRIVER_DIR})
  target_link_options(audiomixer-native PRIVATE -static)
endif()

enable_testing()
add_executable(engine_test test/engine_test.cpp)
target_link_libraries(engine_test audiomixer_engine)
add_test(NAME vectors COMMAND engine_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/vectors.txt)
add_executable(runtime_test test/runtime_test.cpp)
target_link_libraries(runtime_test audiomixer_runtime)
add_test(NAME runtime COMMAND runtime_test)
if(HAVE_RESONANCE)
  add_executable(spatial_test test/spatial_test.cpp)
  target_link_libraries(spatial_test audiomixer_runtime)
  add_test(NAME spatial COMMAND spatial_test)
endif()
