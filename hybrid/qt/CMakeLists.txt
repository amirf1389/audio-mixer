cmake_minimum_required(VERSION 3.21)
project(amfb_hybrid_qt CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2)
qt_add_executable(amfb-hybrid-ui main.cpp MixerBridge.cpp MixerBridge.h)
qt_add_resources(amfb-hybrid-ui "hybrid_qml" PREFIX "/hybrid" FILES Main.qml MixerStrip.qml)
target_link_libraries(amfb-hybrid-ui PRIVATE Qt6::Quick Qt6::QuickControls2)
if(UNIX AND NOT APPLE)
  target_link_libraries(amfb-hybrid-ui PRIVATE rt)
endif()
