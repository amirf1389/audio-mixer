# The hybrid mixer: built from engine/cpp (add_subdirectory), reuses its audio backends.
add_library(amfb_hybrid_ext STATIC engine/main.cpp engine/app_capture.cpp)
target_include_directories(amfb_hybrid_ext PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../engine/cpp/app)
target_link_libraries(amfb_hybrid_ext PUBLIC audiomixer_runtime)
set_target_properties(amfb_hybrid_ext PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(amfb_hybrid SHARED ${CMAKE_CURRENT_SOURCE_DIR}/../engine/cpp/app/ext/ext_shared.cpp)
target_link_libraries(amfb_hybrid PRIVATE amfb_hybrid_ext)
set_target_properties(amfb_hybrid PROPERTIES OUTPUT_NAME "AmF X1 Hybrid" PREFIX "")
amfb_win_version(amfb_hybrid "AmF X1 Hybrid.dll" "AmF X1 Hybrid mixer extension (loaded by the console)" DLL)
add_executable(amfb-hybrid ${CMAKE_CURRENT_SOURCE_DIR}/../engine/cpp/app/ext/ext_main.cpp)
amfb_win_version(amfb-hybrid "AmF X1 Hybrid.exe" "AmF X1 Hybrid mixer" APP)
target_link_libraries(amfb-hybrid PRIVATE amfb_hybrid_ext)
if(WIN32)
  target_link_libraries(amfb_hybrid_ext PUBLIC ole32 avrt)
  if(MINGW)
    foreach(t amfb-hybrid amfb_hybrid)
      target_link_libraries(${t} PRIVATE -Wl,-Bstatic,-lstdc++,-lwinpthread)
    endforeach()
  endif()
elseif(UNIX AND NOT APPLE)
  target_link_libraries(amfb_hybrid_ext PUBLIC rt)
endif()
add_executable(hybrid_core_test test/core_test.cpp)
find_package(Threads REQUIRED)
target_link_libraries(hybrid_core_test PRIVATE Threads::Threads)
add_test(NAME hybrid_core COMMAND hybrid_core_test)
