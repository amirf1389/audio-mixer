# The hybrid mixer: built from engine/cpp (add_subdirectory), reuses its audio backends.
add_executable(amfb-hybrid engine/main.cpp engine/app_capture.cpp)
target_include_directories(amfb-hybrid PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(amfb-hybrid PRIVATE audiomixer_runtime)
if(WIN32)
  target_link_libraries(amfb-hybrid PRIVATE ole32 avrt)
  if(MINGW)
    target_link_libraries(amfb-hybrid PRIVATE -Wl,-Bstatic,-lstdc++,-lwinpthread)
  endif()
elseif(UNIX AND NOT APPLE)
  target_link_libraries(amfb-hybrid PRIVATE rt)
endif()
add_executable(hybrid_core_test test/core_test.cpp)
find_package(Threads REQUIRED)
target_link_libraries(hybrid_core_test PRIVATE Threads::Threads)
add_test(NAME hybrid_core COMMAND hybrid_core_test)
